<title>ブロック崩し（スマホ対応）</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
    background: black;
  }
  canvas {
    display: block;
  }
</style>

<body>

<canvas id="gameCanvas"></canvas>

<script>
const canvas = document.getElementById("gameCanvas");
const ctx = canvas.getContext("2d");

// 画面サイズに合わせる
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

// 背景画像
const bg = new Image();
bg.src = "bg.jpg";

// ボール
let x = canvas.width / 2;
let y = canvas.height - 60;
let dx = 4;
let dy = -4;
let ballRadius = 8;

// パドル
let paddleHeight = 12;
let paddleWidth = canvas.width * 0.25;
let paddleX = (canvas.width - paddleWidth) / 2;

// ブロック
let brickRowCount = 5;
let brickColumnCount = 6;
let brickWidth = canvas.width / 7;
let brickHeight = 20;
let brickPadding = 10;
let brickOffsetTop = 60;
let brickOffsetLeft = 10;

let bricks = [];
for (let c = 0; c < brickColumnCount; c++) {
  bricks[c] = [];
  for (let r = 0; r < brickRowCount; r++) {
    bricks[c][r] = { x: 0, y: 0, status: 1 };
  }
}

// タッチ操作
canvas.addEventListener("touchmove", function(e) {
  let touchX = e.touches[0].clientX;
  paddleX = touchX - paddleWidth / 2;
});

// マウス操作（PCでも使える）
canvas.addEventListener("mousemove", function(e) {
  let relativeX = e.clientX;
  paddleX = relativeX - paddleWidth / 2;
});

// 当たり判定
function collisionDetection() {
  for (let c = 0; c < brickColumnCount; c++) {
    for (let r = 0; r < brickRowCount; r++) {
      let b = bricks[c][r];
      if (b.status === 1) {
        if (x > b.x && x < b.x + brickWidth &&
            y > b.y && y < b.y + brickHeight) {
          dy = -dy;
          b.status = 0;
        }
      }
    }
  }
}

// 描画
function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // 背景
  ctx.drawImage(bg, 0, 0, canvas.width, canvas.height);

  // ボール
  ctx.beginPath();
  ctx.arc(x, y, ballRadius, 0, Math.PI * 2);
  ctx.fillStyle = "white";
  ctx.fill();
  ctx.closePath();

  // パドル
  ctx.fillStyle = "white";
  ctx.fillRect(paddleX, canvas.height - 40, paddleWidth, paddleHeight);

  // ブロック
  for (let c = 0; c < brickColumnCount; c++) {
    for (let r = 0; r < brickRowCount; r++) {
      if (bricks[c][r].status === 1) {
        let brickX = c * (brickWidth + brickPadding) + brickOffsetLeft;
        let brickY = r * (brickHeight + brickPadding) + brickOffsetTop;
        bricks[c][r].x = brickX;
        bricks[c][r].y = brickY;

        ctx.fillStyle = "rgba(255,255,255,0.8)";
        ctx.fillRect(brickX, brickY, brickWidth, brickHeight);
      }
    }
  }

  collisionDetection();

  // 壁反射
  if (x + dx > canvas.width - ballRadius || x + dx < ballRadius) dx = -dx;
  if (y + dy < ballRadius) dy = -dy;
  else if (y + dy > canvas.height - 50) {
    if (x > paddleX && x < paddleX + paddleWidth) {
      dy = -dy;
    } else {
      alert("ゲームオーバー");
      document.location.reload();
    }
  }

  x += dx;
  y += dy;

  requestAnimationFrame(draw);
}

// 開始
bg.onload = () => draw();
</script>

</body>
